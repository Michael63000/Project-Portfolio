#include <VarSpeedServo.h> 
VarSpeedServo servo1;
int algCounter = 0; //time the algorithm

//Exponential smoothing
float s_i = 0; //smoothened data point for output to serial
float s_i_prev = 0; //previous smoothened value
float a = 0.5; //smoothing factor {0 < a < 1}, gives lowest MSE (mean square error)

//Servo driver
int MAX_VALUE = 151; //Max of 3 full strength EMG measurements
int MAX_ANGLE = 135; //Max angle the servo will turn (fully closed hand)
int MIN_ANGLE = 180;
float normValue = 0.0; //Normalized value
int mapped_value = 0; //Mapped value norm->speed
bool handOpen = true; //Hand is open initially

void setup(){
  Serial.begin(9600);
  servo1.attach(A1); //analog pin A1
  servo1.slowmove(MIN_ANGLE, 150); //servo to position 0
}
void loop(){
  int sensorValue = analogRead(A0); //sensor to analog pin A0
 if(algCounter == 0){ //initial i = 0
   s_i = sensorValue; //smoothened value is raw data value for first point i = 0
   Serial.println(s_i); //output the i = 0 value
   s_i_prev = s_i; //save s_i in s_(i-1)
 }
 else{ //i > 0
   s_i = a * sensorValue + (1 - a) * s_i_prev; //smoothened value eqn for i > 0
   Serial.println(s_i); //output the smoothened value
   s_i_prev = s_i; //save s_i for the i + 1 iteration
 }
  algCounter++; 

  normValue = (float) s_i / MAX_VALUE; //find normalized value from MVIC
  if(normValue > 1.0){
    normValue = 1.0; //restrict normalized value 0.0 < x < 1.0
  }
  mapped_value = 1 + normValue * 199; //Map the normalized value between [0,1] to [1,200] for speed control
  // thresholds with hysteresis
  if(normValue > 0.33 && handOpen){ //if hand status is open
    servo1.slowmove(MAX_ANGLE, mapped_value);
    handOpen = false; //Hand is now closed
  }
  if(normValue < 0.13 && !handOpen){
    servo1.slowmove(MIN_ANGLE, 200); //Open the hand at constant rate (max speed)
    handOpen = true; //Hand is now open
  }
  delay(20); //delay in between reads for stability
}
